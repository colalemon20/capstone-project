#!/usr/bin/env python3
import time, serial, serial.tools.list_ports
import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, DurabilityPolicy, HistoryPolicy
from geometry_msgs.msg import Twist

def clamp(x, lo, hi): return max(lo, min(hi, x))

class SerialDiffDriveBridge(Node):
    def __init__(self):
        super().__init__('serial_diff_drive_bridge')
        self.declare_parameter('serial_port', '/dev/ttyACM0')
        self.declare_parameter('baudrate', 115200)
        self.declare_parameter('max_v', 0.30)
        self.declare_parameter('max_w', 2.00)
        self.declare_parameter('cmd_timeout', 0.5)
        self.declare_parameter('frame_format', '<{:+.3f},{:+.3f}>\n')
        self.declare_parameter('echo_interval', 0.5)

        self.port = self.get_parameter('serial_port').value
        self.baudrate = int(self.get_parameter('baudrate').value)
        self.max_v = float(self.get_parameter('max_v').value)
        self.max_w = float(self.get_parameter('max_w').value)
        self.cmd_timeout = float(self.get_parameter('cmd_timeout').value)
        self.frame_format = str(self.get_parameter('frame_format').value)
        self.echo_interval = float(self.get_parameter('echo_interval').value)

        self.ser = None
        self._rx_buf = bytearray()
        self._last_rx_line = None
        self._last_echo_ts = time.monotonic()
        self.last_cmd_time = time.monotonic() - 999.0
        self.last_tx_stop = True

        qos = QoSProfile(reliability=ReliabilityPolicy.RELIABLE,
                         durability=DurabilityPolicy.VOLATILE,
                         history=HistoryPolicy.KEEP_LAST, depth=10)
        self.sub = self.create_subscription(Twist, '/cmd_vel', self.cb_cmd_vel, qos)
        self.create_timer(0.01, self.on_timer)
        self.create_timer(1.0, self.try_open_serial)

    def try_open_serial(self):
        if self.ser and self.ser.is_open: return
        try:
            self.ser = serial.Serial(self.port, self.baudrate, timeout=0.0)
            self.get_logger().info(f'Open {self.port} @{self.baudrate}')
            self._rx_buf.clear(); self._last_rx_line=None; self._last_echo_ts=time.monotonic()
        except serial.SerialException as e:
            self.get_logger().warn(f'Failed to open {self.port}: {e}')

    def send_frame(self, v_norm, w_norm):
        if not (self.ser and self.ser.is_open): return
        try: self.ser.write(self.frame_format.format(v_norm, w_norm).encode('ascii'))
        except serial.SerialException as e:
            self.get_logger().warn(f'Serial write error: {e}')
            try: self.ser.close()
            except: pass

    def drain_rx(self):
        if not (self.ser and self.ser.is_open): return
        try: n = self.ser.in_waiting
        except serial.SerialException: return
        if not n: return
        try: chunk = self.ser.read(n)
        except serial.SerialException: return
        if not chunk: return
        self._rx_buf.extend(chunk)
        while True:
            nl = self._rx_buf.find(b'\n')
            if nl < 0: break
            raw = self._rx_buf[:nl]; self._rx_buf = self._rx_buf[nl+1:]
            s = raw.rstrip(b'\r').decode('ascii', errors='ignore').strip()
            if s: self._last_rx_line = s

    def cb_cmd_vel(self, msg: Twist):
        v_norm = clamp(msg.linear.x / self.max_v, -1.0, 1.0)
        w_norm = clamp(msg.angular.z / self.max_w, -1.0, 1.0)
        self.send_frame(v_norm, w_norm)
        self.last_cmd_time = time.monotonic()
        self.last_tx_stop = (abs(v_norm)<1e-6 and abs(w_norm)<1e-6)

    def on_timer(self):
        self.drain_rx()
        now = time.monotonic()
        if self._last_rx_line and (now - self._last_echo_ts) >= self.echo_interval:
            self.get_logger().info(f'[ARD] {self._last_rx_line}')
            self._last_rx_line = None
            self._last_echo_ts = now
        if (now - self.last_cmd_time) > self.cmd_timeout and not self.last_tx_stop:
            self.send_frame(0.0, 0.0); self.last_tx_stop = True

def main(args=None):
    rclpy.init(args=args)
    node = SerialDiffDriveBridge()
    try: rclpy.spin(node)
    except KeyboardInterrupt: pass
    finally:
        if node.ser and node.ser.is_open:
            try: node.send_frame(0.0, 0.0); node.ser.close()
            except: pass
        node.destroy_node(); rclpy.shutdown()

if __name__ == '__main__':
    main()
