#include <math.h> 
/********************************************************************
 * Diff-Drive with Encoder PID (Arduino UNO/Nano)
 * - 시리얼 수신: "<v_norm,w_norm>\n"  (예: <+0.300,+0.000>)
 * - v_norm ∈ [-1,1]  → v = v_norm * MAX_V     [m/s]
 * - w_norm ∈ [-1,1]  → w = w_norm * MAX_W     [rad/s]
 * - 좌/우 목표속도: vL = v - w*B/2, vR = v + w*B/2
 * - 각 바퀴 PID로 목표 RPM 추종 + 직진시 동기화 보정
 ********************************************************************/

// ── Motor pins (변경 없이 사용) ───────────────────────────────
const byte L_PWM  = 10, L_CW  =  13, L_CCW = 6;
const byte R_PWM  =  9, R_CW  = 12, R_CCW =  5;

// ── Encoder pins (A는 2/3: 외부인터럽트) ─────────────────────
const byte ENC_L_A = 3, ENC_L_B = 7;
const byte ENC_R_A = 2, ENC_R_B = 4;

// ── Kinematics / Encoder params ───────────────────────────────
const float WHEEL_RADIUS = 0.03f;          // [m]  (r=3cm)
const float WHEEL_CIRC   = 2.0f * PI * WHEEL_RADIUS;  // ≈ 0.188496 m
const float WHEEL_BASE   = 0.30f;          // [m]  축간 거리(필요시 조정)
const long  CPR          = 9600;           // counts per revolution


// ── Command scaling (라즈베리파이 브리지와 맞춰야 함) ────────
const float MAX_V = 0.30f;   // [m/s]
const float MAX_W = 2.00f;   // [rad/s]

// ── Control loop timing ───────────────────────────────────────
const unsigned long CTRL_DT_MS = 20;   // 50 Hz 제어
const unsigned long PRINT_DT_MS = 100; // 10 Hz 디버그 출력

// ── Motor drive config ────────────────────────────────────────
const int PWM_MIN = 30;   // 정지 마찰 극복용 최소 듀티(0~255, 드라이버에 맞춰 조정)
const int PWM_MAX = 255;

// ── Motor polarity (전진 기준 부호) ───────────────────────────
const int MOTOR_L_SIGN = -1;  // 왼쪽
const int MOTOR_R_SIGN = +1;  // 오른쪽

// ── Encoder sign (배선/정의에 따라 회전부호가 반대로 나오면 ±1 바꾸기) ─
const int ENC_L_SIGN = -1;
const int ENC_R_SIGN = +1;

// ── Straight sync (직진 시 L/R RPM 동기화 보정) ────────────────
const float SYNC_EPS_W = 0.05f;     // |ω|<EPS면 직진으로 간주
const float K_SYNC     = 0.10f;     // RPM 차이 보정 게인 (0~0.3부터 조정)

// ── PID gains (초기 값, 이후 주행하며 조정 권장) ───────────────
struct PID {
  float kp, ki, kd;
  float i, prev_e;
  float out;
};
PID pidL{ 0.80f, 2.00f, 0.00f, 0.0f, 0.0f, 0.0f };
PID pidR{ 0.80f, 2.00f, 0.00f, 0.0f, 0.0f, 0.0f };

// Anti-windup용 적분 상한(대략적인 PWM 범위를 RPM으로 환산해 제한)
const float I_CLAMP = 200.0f; // 적분항 절대 최대치

// ── State ─────────────────────────────────────────────────────
volatile long encL = 0, encR = 0;   // 누적 카운트
long prevEncL = 0, prevEncR = 0;
unsigned long lastCtrl = 0, lastPrint = 0;

float cmd_v = 0.0f, cmd_w = 0.0f;     // [m/s], [rad/s]
float tgt_rpm_L = 0.0f, tgt_rpm_R = 0.0f;
float rpmL = 0.0f, rpmR = 0.0f;

unsigned long lastCmdTime = 0;        // 명령 수신 타임스탬프
const unsigned long CMD_TIMEOUT_MS = 600;  // 무명령시 정지

// ── Utils ─────────────────────────────────────────────────────
inline int clampi(int x, int lo, int hi){ return (x<lo)?lo:((x>hi)?hi:x); }
inline float clampf(float x, float lo, float hi){ return (x<lo)?lo:((x>hi)?hi:x); }

// PWM 출력(방향 포함). speed_pwm≥0 가정, dirSign: +1=정방향 / -1=역방향
void driveMotor(byte pinPWM, byte pinCW, byte pinCCW, int dirSign, int speed_pwm) {
  int pwm = clampi(speed_pwm, 0, PWM_MAX);
  if (pwm == 0) {
    digitalWrite(pinCW, LOW);
    digitalWrite(pinCCW, LOW);
    analogWrite(pinPWM, 0);
    return;
  }
  // 마찰 극복을 위한 오프셋
  pwm = PWM_MIN + (int)((long)(pwm) * (PWM_MAX - PWM_MIN) / PWM_MAX);

  if (dirSign >= 0) {
    digitalWrite(pinCW, HIGH);
    digitalWrite(pinCCW, LOW);
  } else {
    digitalWrite(pinCW, LOW);
    digitalWrite(pinCCW, HIGH);
  }
  analogWrite(pinPWM, pwm);
}

// PID 업데이트: set(=목표RPM), mea(=실제RPM), dt(초)
float pid_update(PID &p, float set, float mea, float dt) {
  float e  = set - mea;
  p.i     += e * p.ki * dt;
  p.i      = clampf(p.i, -I_CLAMP, I_CLAMP);
  float d  = (e - p.prev_e) / dt;
  p.out    = p.kp * e + p.i + p.kd * d;
  p.prev_e = e;
  return p.out;
}

// ── Encoders ISR (쿼드러처: A변화 시 B로 방향판별) ─────────────
void isrLeft(){
  bool a = digitalRead(ENC_L_A);
  bool b = digitalRead(ENC_L_B);
  encL += ((a == b) ? +1 : -1) * ENC_L_SIGN;
}
void isrRight(){
  bool a = digitalRead(ENC_R_A);
  bool b = digitalRead(ENC_R_B);
  encR += ((a == b) ? +1 : -1) * ENC_R_SIGN;
}

// ── Serial parsing "<v,w>\n" ───────────────────────────────────
void parseSerial() {
  static char buf[40]; static int idx = 0;
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '<') { idx = 0; }
    if (idx < (int)sizeof(buf)-1) buf[idx++] = c;
    if (c == '\n') {
      buf[idx] = '\0';
      float v_norm, w_norm;
      if (sscanf(buf, "<%f,%f>", &v_norm, &w_norm) == 2) {
        v_norm = clampf(v_norm, -1.0f, 1.0f);
        w_norm = clampf(w_norm, -1.0f, 1.0f);
        cmd_v  = v_norm * MAX_V;
        cmd_w  = w_norm * MAX_W;
        lastCmdTime = millis();
      }
      idx = 0;
    }
  }
}

// ── Setup ──────────────────────────────────────────────────────
void setup(){
  // Motor pins
  pinMode(L_PWM, OUTPUT); pinMode(L_CW, OUTPUT); pinMode(L_CCW, OUTPUT);
  pinMode(R_PWM, OUTPUT); pinMode(R_CW, OUTPUT); pinMode(R_CCW, OUTPUT);
  // Stop all
  driveMotor(L_PWM, L_CW, L_CCW, +1, 0);
  driveMotor(R_PWM, R_CW, R_CCW, +1, 0);

  // Encoders
  pinMode(ENC_L_A, INPUT_PULLUP); pinMode(ENC_L_B, INPUT_PULLUP);
  pinMode(ENC_R_A, INPUT_PULLUP); pinMode(ENC_R_B, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(ENC_L_A), isrLeft,  CHANGE);
  attachInterrupt(digitalPinToInterrupt(ENC_R_A), isrRight, CHANGE);

  // Serial
  Serial.begin(115200);
  while(!Serial) { ; }
  Serial.println(F("\n[DiffDrive-PID] start"));
  Serial.println(F("Expect frames: <v_norm,w_norm> (e.g., <+0.300,+0.000>)"));

  lastCmdTime = millis();
  lastCtrl = lastPrint = millis();
}

// ── Main loop ─────────────────────────────────────────────────
void loop(){
  // 1) Receive command
  parseSerial();

  // 2) Command timeout → stop
  if (millis() - lastCmdTime > CMD_TIMEOUT_MS) {
    cmd_v = 0.0f; cmd_w = 0.0f;
  }

  // 3) Control loop (50 Hz)
  unsigned long now = millis();
  if (now - lastCtrl >= CTRL_DT_MS) {
    float dt = (now - lastCtrl) / 1000.0f;
    lastCtrl = now;

    // 목표 좌/우 선속도 [m/s]
    float vL = cmd_v - cmd_w * (WHEEL_BASE * 0.5f);
    float vR = cmd_v + cmd_w * (WHEEL_BASE * 0.5f);

    // 목표 RPM 변환: rev/s = v/C → RPM = v/C*60
    tgt_rpm_L = (vL / WHEEL_CIRC) * 60.0f;
    tgt_rpm_R = (vR / WHEEL_CIRC) * 60.0f;

    // 실제 RPM 측정 (Δcount → RPM)
    long curL, curR;
    noInterrupts(); curL = encL; curR = encR; interrupts();
    long dN_L = curL - prevEncL; prevEncL = curL;
    long dN_R = curR - prevEncR; prevEncR = curR;

    rpmL = (float)dN_L * 60.0f / (CPR * dt);
    rpmR = (float)dN_R * 60.0f / (CPR * dt);

    // 직진시 동기화 보정(옵션): 좌/우 RPM 차이를 줄이도록 목표에 소량 가감
    if (fabsf(cmd_w) < SYNC_EPS_W) {
      float diff = rpmR - rpmL; // R-L 양수면 오른쪽이 더 빠름
      tgt_rpm_L += K_SYNC * diff;
      tgt_rpm_R -= K_SYNC * diff;
    }

    // 휠별 PID → 제어출력(임시 "PWM 등가" 값, sign 포함)
    float uL = pid_update(pidL, tgt_rpm_L, rpmL, dt);
    float uR = pid_update(pidR, tgt_rpm_R, rpmR, dt);

    // 방향/크기 분리 후 PWM 변환
    int dirL = ((uL >= 0) ? +1 : -1) * MOTOR_L_SIGN;
    int dirR = ((uR >= 0) ? +1 : -1) * MOTOR_R_SIGN;

    // 간단 선형 맵핑: RPM 오차를 PWM으로 (게인 조절)
    // (필요 시 feedforward 추가 가능: pwm_base = k_ff * |tgt_rpm|)
    const float K_PWM = 2.0f; // RPM→PWM 스케일(초기값, 주행하며 조정)
    int pwmL = (int)fabs(uL * K_PWM);
    int pwmR = (int)fabs(uR * K_PWM);

    // Saturation
    pwmL = clampi(pwmL, 0, PWM_MAX);
    pwmR = clampi(pwmR, 0, PWM_MAX);

    // 정지 명령이면 바로 모터 off
    if (cmd_v == 0.0f && cmd_w == 0.0f) { pwmL = 0; pwmR = 0; }

    // 실제 모터 구동
    driveMotor(L_PWM, L_CW, L_CCW, dirL, pwmL);
    driveMotor(R_PWM, R_CW, R_CCW, dirR, pwmR);
  }

  // 4) Debug print (10 Hz)
  if (millis() - lastPrint >= PRINT_DT_MS) {
    lastPrint = millis();
    Serial.print(F("v=")); Serial.print(cmd_v,3);
    Serial.print(F(" m/s  w=")); Serial.print(cmd_w,3);
    Serial.print(F(" rad/s | "));
    Serial.print(F("RPM L tgt/act=")); Serial.print(tgt_rpm_L,1); Serial.print('/'); Serial.print(rpmL,1);
    Serial.print(F("  R ")); Serial.print(tgt_rpm_R,1); Serial.print('/'); Serial.print(rpmR,1);
    Serial.println();
  }
}
