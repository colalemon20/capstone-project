#!/usr/bin/env python3
# -*- coding: utf-8 -*-
import re, time, math, serial
import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy, DurabilityPolicy
from nav_msgs.msg import Odometry
from geometry_msgs.msg import TransformStamped
from tf2_ros import TransformBroadcaster

RPM_LINE = re.compile(r'rpmL\s*=\s*([\-0-9\.]+).*rpmR\s*=\s*([\-0-9\.]+)')

def clamp(x, lo, hi): return max(lo, min(hi, x))

class WheelOdomPublisher(Node):
    def __init__(self):
        super().__init__('wheel_odom_publisher')

        # ---- params ----
        self.declare_parameter('serial_port', '/dev/ttyACM0')
        self.declare_parameter('baudrate', 115200)
        self.declare_parameter('radius', 0.03)         # m
        self.declare_parameter('wheel_base', 0.30)     # m
        self.declare_parameter('frame_id', 'odom')
        self.declare_parameter('child_frame_id', 'base_link')
        self.declare_parameter('topic', '/odom')
        self.declare_parameter('publish_tf', True)

        self.port = self.get_parameter('serial_port').value
        self.baud = int(self.get_parameter('baudrate').value)
        self.r = float(self.get_parameter('radius').value)
        self.B = float(self.get_parameter('wheel_base').value)
        self.frame_id = self.get_parameter('frame_id').value
        self.child_frame_id = self.get_parameter('child_frame_id').value
        self.topic = self.get_parameter('topic').value
        self.pub_tf = bool(self.get_parameter('publish_tf').value)

        # ---- serial ----
        self.ser = None
        self._buf = bytearray()

        # ---- odom state ----
        self.x = 0.0
        self.y = 0.0
        self.th = 0.0
        self.last_t = time.monotonic()

        # precompute
        self.K = (2.0 * math.pi * self.r) / 60.0  # RPM -> m/s 怨꾩닔

        # publishers
        qos = QoSProfile(
            reliability=ReliabilityPolicy.BEST_EFFORT,
            durability=DurabilityPolicy.VOLATILE,
            history=HistoryPolicy.KEEP_LAST, depth=10
        )
        self.pub = self.create_publisher(Odometry, self.topic, qos)
        self.tfb = TransformBroadcaster(self)

        # timers
        self.create_timer(0.01, self._on_timer)      # 100 Hz read/compute
        self.create_timer(1.0, self._try_open_serial)

        self.get_logger().info(f'wheel_odom -> {self.topic} (frame={self.frame_id}, child={self.child_frame_id})')

        # simple covariances (?꾩슂??議곗젙)
        self.pose_cov = [0.0]*36
        self.twist_cov = [0.0]*36
        # x,y,theta留??곷떦???좊ː: ?섎㉧吏???ш쾶
        self.pose_cov[0]  = 1e-3   # x
        self.pose_cov[7]  = 1e-3   # y
        self.pose_cov[35] = 1e-2   # yaw
        self.pose_cov[14] = 1e6    # z
        self.pose_cov[21] = 1e6    # roll
        self.pose_cov[28] = 1e6    # pitch

        self.twist_cov[0]  = 1e-2  # vx
        self.twist_cov[7]  = 1e-2  # vy
        self.twist_cov[35] = 1e-2  # wz
        self.twist_cov[14] = 1e6
        self.twist_cov[21] = 1e6
        self.twist_cov[28] = 1e6

    def _try_open_serial(self):
        if self.ser and self.ser.is_open:
            return
        try:
            self.ser = serial.Serial(self.port, self.baud, timeout=0.0)
            self.get_logger().info(f'Open {self.port} @{self.baud}')
            self._buf.clear()
        except serial.SerialException as e:
            self.get_logger().warn(f'Serial open failed: {e}')

    def _drain(self):
        if not (self.ser and self.ser.is_open):
            return
        try:
            n = self.ser.in_waiting
        except serial.SerialException:
            return
        if not n: return
        try:
            chunk = self.ser.read(n)
        except serial.SerialException:
            return
        if not chunk: return
        self._buf.extend(chunk)
        while True:
            i = self._buf.find(b'\n')
            if i < 0:
                break
            raw = self._buf[:i]
            self._buf = self._buf[i+1:]
            line = raw.rstrip(b'\r').decode('ascii', errors='ignore')
            self._handle_line(line)

    def _handle_line(self, s: str):
        # 湲곕? ?쇱씤 ?? "rpmL=100.75 rpmR=99.20 diff=... ..."
        m = RPM_LINE.search(s)
        if not m:
            return
        try:
            rpmL = float(m.group(1))
            rpmR = float(m.group(2))
        except ValueError:
            return

        now = time.monotonic()
        dt = now - self.last_t
        if dt <= 0.0 or dt > 1.0:
            self.last_t = now
            return

        v = 0.5 * (rpmL + rpmR) * self.K            # m/s
        w = ((rpmR - rpmL) * self.K) / self.B       # rad/s

        # ?곷텇
        self.th += w * dt
        c, s = math.cos(self.th), math.sin(self.th)
        self.x += v * c * dt
        self.y += v * s * dt
        self.last_t = now

        # ?쇰툝由ъ떆
        self._publish(now, v, w)

    def _publish(self, now_mono: float, v: float, w: float):
        stamp = self.get_clock().now().to_msg()

        # Odometry
        od = Odometry()
        od.header.stamp = stamp
        od.header.frame_id = self.frame_id
        od.child_frame_id = self.child_frame_id

        od.pose.pose.position.x = self.x
        od.pose.pose.position.y = self.y
        od.pose.pose.position.z = 0.0
        qz = math.sin(self.th * 0.5)
        qw = math.cos(self.th * 0.5)
        od.pose.pose.orientation.z = qz
        od.pose.pose.orientation.w = qw
        od.pose.covariance = self.pose_cov

        od.twist.twist.linear.x = v
        od.twist.twist.linear.y = 0.0
        od.twist.twist.angular.z = w
        od.twist.covariance = self.twist_cov

        self.pub.publish(od)

        # TF odom->base_link
        if self.pub_tf:
            tf = TransformStamped()
            tf.header.stamp = stamp
            tf.header.frame_id = self.frame_id
            tf.child_frame_id = self.child_frame_id
            tf.transform.translation.x = self.x
            tf.transform.translation.y = self.y
            tf.transform.translation.z = 0.0
            tf.transform.rotation.z = qz
            tf.transform.rotation.w = qw
            self.tfb.sendTransform(tf)

    def _on_timer(self):
        self._drain()

def main():
    rclpy.init()
    node = WheelOdomPublisher()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        if node.ser and node.ser.is_open:
            try: node.ser.close()
            except: pass
        node.destroy_node()
        rclpy.shutdown()

if __name__ == '__main__':
    main()
